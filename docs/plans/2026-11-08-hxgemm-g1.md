# hxgemm G1 (FP32 SGEMM 단계별) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** naive부터 cp.async 2·3단 파이프라인까지 FP32 SGEMM 커널 5단계를 만들고, cuBLAS와 같은 조건에서 재서 마지막 커널이 4096³에서 cuBLAS 중앙값 속도의 80% 이상인지 판정하고, 단계마다 Nsight 근거를 남긴다.

**Architecture:** 커널은 `csrc/kernels/*.cu`, 공개 진입점은 C 함수 `hxgemm_sgemm`(인자 검사 → 커널 선택). C++ 벤치가 검증과 시간 측정을 맡고 `ncu` 대상이 된다. PyTorch 바인딩은 vLLM `csrc/torch_bindings.cpp` 방식으로 등록하고 pytest가 `torch.matmul`과 비교한다. 코드는 맥에서 쓰고 `rsync`로 데스크탑 WSL2에 보내 SSH로 빌드·실행한다.

**Tech Stack:** CUDA C++17(sm_86), CMake ≥ 3.24, cuBLAS, PyTorch C++ extension, pytest, Nsight Compute CLI. 버전은 Task 0에서 기록한다.

**Spec:** `docs/specs/2026-11-08-hxgemm-g1-design.md`

## Global Constraints

- 행 우선 FP32, C = αAB + βC. β = 0이면 C를 읽지 않는다(BLAS 의미론, C에 NaN이 있어도 결과에 퍼지지 않음).
- M, N, K 중 하나가 0이면 아무것도 하지 않음. K = 0이면 C = βC.
- 반환: 0 성공, −1 인자 오류(커널 실행 전), −2 지원하지 않는 타일, 양수는 `cudaError_t`.
- 측정: warm-up 10, 측정 50(각각 CUDA event), 중앙값·최소·최대. 매 측정 전 L2 × 2 버퍼 쓰기로 L2 비우기.
- 공정성: cuBLAS `CUBLAS_DEFAULT_MATH`, PyTorch `allow_tf32 = False`, 환경 `NVIDIA_TF32_OVERRIDE=0`.
- 허용 오차: |x − ref| ≤ atol + rtol·|ref|, rtol = 1e−5·√K, atol = 1e−6·K. Task 2에서 cuBLAS 자신이 FP64 대비 이 기준을 통과하는지 먼저 확인한다.
- 원격 실행: `rsync -az --delete --exclude build --exclude .superpowers ~/Desktop/hxgemm/ desk:~/hxgemm/` 후 `ssh desk 'cd ~/hxgemm && <cmd>'`. 이하 "원격:"은 이 두 단계를 뜻한다.

## Review Focus

1. **β = 0이고 C에 NaN이 들어 있는 입력**: 결과에 NaN이 없어야 한다 → Task 3 `test_beta_zero_ignores_nan`(C++ 벤치 검증 모드)와 Task 8 pytest.
2. **정렬이 안 맞는 포인터(오프셋 1 float)와 ld > 열 수**: float4/cp.async 16 B 경로가 아니라 스칼라 경로로 맞게 계산 → Task 5·6 검증 목록, Task 8 pytest.
3. **M = 1, N = 1 같은 아주 작은 모양과 타일보다 작은 K**: 경계 처리 → 모든 커널 검증 목록에 1, 7, 33 포함.
4. **shared memory 한도를 넘는 타일 조합 요청**: −2 반환, 크래시 없음 → Task 7 `unsupported_tile_rejected`.
5. **GPU 클럭 변동으로 인한 측정 흔들림**: 클럭 로그 없이 숫자를 비교하지 않음 → Task 9 측정 스크립트가 클럭 로그를 같이 저장하지 않으면 실패.

```cpp
#include "gemm.h"
#include "kernels/common.cuh"
#include "kernels/launch.h"

namespace {
bool bad_args(int M, int N, int K, const float* A, int lda, const float* B, int ldb,
              const float* C, int ldc) {
  if (M < 0 || N < 0 || K < 0) return true;
  if (M == 0 || N == 0) return false;
  if (ldc < N || !C) return true;
  if (K == 0) return false;
  return !A || !B || lda < K || ldb < N;
}
}  // namespace

extern "C" int hxgemm_sgemm(hxgemm_kernel_t k, hxgemm_tile_t t, int M, int N, int K,
                            float alpha, const float* A, int lda, const float* B, int ldb,
                            float beta, float* C, int ldc, cudaStream_t s) {
  if (k < HXGEMM_K01 || k > HXGEMM_K05) return HXGEMM_EINVAL;
  if (bad_args(M, N, K, A, lda, B, ldb, C, ldc)) return HXGEMM_EINVAL;
  if (M == 0 || N == 0) return HXGEMM_OK;
  if (K == 0) {
    dim3 b(32, 8), g((N + 31) / 32, (M + 7) / 8);
    scale_c<<<g, b, 0, s>>>(M, N, beta, C, ldc);
    return static_cast<int>(cudaGetLastError());
  }
  int r = hx_launch(k, t, M, N, K, alpha, A, lda, B, ldb, beta, C, ldc, s);
  if (r != HXGEMM_OK) return r;
  return static_cast<int>(cudaGetLastError());
}
```

```cpp
#pragma once
#include <algorithm>
#include <cmath>
#include <cstdio>
#include <functional>
#include <random>
#include <vector>
#include <cuda_runtime.h>

#define CK(x) do { cudaError_t e_ = (x); if (e_ != cudaSuccess) { \
  std::fprintf(stderr, "%s:%d %s\n", __FILE__, __LINE__, cudaGetErrorString(e_)); std::exit(2); } } while (0)

inline void fill_uniform(std::vector<float>& v, unsigned seed) {
  std::mt19937 g(seed);
  std::uniform_real_distribution<float> d(-1.f, 1.f);
  for (auto& x : v) x = d(g);
}

// row-major, C = alpha*A*B + beta*C0 in double
inline void ref_cpu_fp64(int M, int N, int K, float alpha, const std::vector<float>& A, int lda,
                         const std::vector<float>& B, int ldb, float beta,
                         const std::vector<float>& C0, int ldc, std::vector<double>& out) {
  out.assign(static_cast<size_t>(M) * ldc, 0.0);
  for (int i = 0; i < M; ++i)
    for (int j = 0; j < N; ++j) {
      double acc = 0;
      for (int k = 0; k < K; ++k) acc += double(A[i * lda + k]) * double(B[k * ldb + j]);
      double c = beta == 0.f ? 0.0 : double(beta) * C0[i * ldc + j];
      out[i * ldc + j] = double(alpha) * acc + c;
    }
}

inline bool check(const float* got, const double* ref, int M, int N, int ldc, int K,
                  double* max_abs, double* max_rel) {
  const double rtol = 1e-5 * std::sqrt(double(K)), atol = 1e-6 * K;
  bool ok = true;
  *max_abs = *max_rel = 0;
  for (int i = 0; i < M; ++i)
    for (int j = 0; j < N; ++j) {
      double r = ref[i * ldc + j], g = got[i * ldc + j];
      double err = std::fabs(g - r);
      if (std::isnan(g) || err > atol + rtol * std::fabs(r)) ok = false;
      *max_abs = std::max(*max_abs, err);
      if (r != 0) *max_rel = std::max(*max_rel, err / std::fabs(r));
    }
  return ok;
}

// Each iteration: flush L2 (write flush_bytes), then time one launch with events.
inline float time_kernel(const std::function<void()>& launch, int warmup, int iters,
                         float* min_us, float* max_us, void* flush_buf, size_t flush_bytes) {
  for (int i = 0; i < warmup; ++i) launch();
  CK(cudaDeviceSynchronize());
  cudaEvent_t a, b;
  CK(cudaEventCreate(&a));
  CK(cudaEventCreate(&b));
  std::vector<float> t(iters);
  for (int i = 0; i < iters; ++i) {
    CK(cudaMemsetAsync(flush_buf, i & 0xff, flush_bytes));
    CK(cudaEventRecord(a));
    launch();
    CK(cudaEventRecord(b));
    CK(cudaEventSynchronize(b));
    float ms = 0;
    CK(cudaEventElapsedTime(&ms, a, b));
    t[i] = ms * 1000.f;
  }
  cudaEventDestroy(a);
  cudaEventDestroy(b);
  std::sort(t.begin(), t.end());
  *min_us = t.front();
  *max_us = t.back();
  return t[iters / 2];
}
```

```cpp
#include "common.cuh"
#include "launch.h"
template <int T>
__global__ void k03_smem(int M, int N, int K, float alpha, const float* A, int lda,
                         const float* B, int ldb, float beta, float* C, int ldc) {
  __shared__ float As[T][T];
  __shared__ float Bs[T][T];
  const int tx = threadIdx.x, ty = threadIdx.y;
  const int row = blockIdx.y * T + ty, col = blockIdx.x * T + tx;
  float acc = 0.f;
  for (int k0 = 0; k0 < K; k0 += T) {
    As[ty][tx] = (row < M && k0 + tx < K) ? A[row * lda + k0 + tx] : 0.f;
    Bs[ty][tx] = (k0 + ty < K && col < N) ? B[(k0 + ty) * ldb + col] : 0.f;
    __syncthreads();
#pragma unroll
    for (int k = 0; k < T; ++k) acc += As[ty][k] * Bs[k][tx];
    __syncthreads();
  }
  if (row < M && col < N)
    C[row * ldc + col] = hx_out(acc, alpha, beta, beta == 0.f ? 0.f : C[row * ldc + col]);
}

  k03_smem<32><<<g, b, 0, s>>>(M, N, K, alpha, A, lda, B, ldb, beta, C, ldc);
}
```

```cpp
#include "common.cuh"
#include "launch.h"
#include "tiles.h"

// Thread computes a TM x TN block of C. A tile stored transposed (As[BK][BM]) so the inner
// loop reads TM consecutive floats; B tile stored as is (Bs[BK][BN]).
template <int BM, int BN, int BK, int TM, int TN, bool VEC>
__global__ void __launch_bounds__((BM / TM) * (BN / TN))
k04_reg(int M, int N, int K, float alpha, const float* A, int lda, const float* B, int ldb,
        float beta, float* C, int ldc) {
  constexpr int NT = (BM / TM) * (BN / TN);
  extern __shared__ float smem[];
  float* As = smem;            // [BK][BM]
  float* Bs = smem + BK * BM;  // [BK][BN]
  const int tid = threadIdx.x;
  const int bm0 = blockIdx.y * BM, bn0 = blockIdx.x * BN;
  const int tr = (tid / (BN / TN)) * TM, tc = (tid % (BN / TN)) * TN;
  float acc[TM][TN] = {};
  float ra[TM], rb[TN];
  for (int k0 = 0; k0 < K; k0 += BK) {
    if (VEC) {
      for (int i = tid; i < BM * BK / 4; i += NT) {
        const int r = i / (BK / 4), c = (i % (BK / 4)) * 4, gr = bm0 + r, gc = k0 + c;
        float4 v = make_float4(0.f, 0.f, 0.f, 0.f);
        if (gr < M && gc + 3 < K) v = *reinterpret_cast<const float4*>(&A[gr * lda + gc]);
        else if (gr < M) {
          float* p = &v.x;
          for (int q = 0; q < 4; ++q) p[q] = gc + q < K ? A[gr * lda + gc + q] : 0.f;
        }
        As[(c + 0) * BM + r] = v.x; As[(c + 1) * BM + r] = v.y;
        As[(c + 2) * BM + r] = v.z; As[(c + 3) * BM + r] = v.w;
      }
      for (int i = tid; i < BK * BN / 4; i += NT) {
        const int r = i / (BN / 4), c = (i % (BN / 4)) * 4, gr = k0 + r, gc = bn0 + c;
        float4 v = make_float4(0.f, 0.f, 0.f, 0.f);
        if (gr < K && gc + 3 < N) v = *reinterpret_cast<const float4*>(&B[gr * ldb + gc]);
        else if (gr < K) {
          float* p = &v.x;
          for (int q = 0; q < 4; ++q) p[q] = gc + q < N ? B[gr * ldb + gc + q] : 0.f;
        }
        *reinterpret_cast<float4*>(&Bs[r * BN + c]) = v;
      }
    } else {
      for (int i = tid; i < BM * BK; i += NT) {
        const int r = i / BK, c = i % BK;
        As[c * BM + r] = (bm0 + r < M && k0 + c < K) ? A[(bm0 + r) * lda + k0 + c] : 0.f;
      }
      for (int i = tid; i < BK * BN; i += NT) {
        const int r = i / BN, c = i % BN;
        Bs[r * BN + c] = (k0 + r < K && bn0 + c < N) ? B[(k0 + r) * ldb + bn0 + c] : 0.f;
      }
    }
    __syncthreads();
#pragma unroll
    for (int kk = 0; kk < BK; ++kk) {
#pragma unroll
      for (int i = 0; i < TM; ++i) ra[i] = As[kk * BM + tr + i];
#pragma unroll
      for (int j = 0; j < TN; ++j) rb[j] = Bs[kk * BN + tc + j];
#pragma unroll
      for (int i = 0; i < TM; ++i)
#pragma unroll
        for (int j = 0; j < TN; ++j) acc[i][j] += ra[i] * rb[j];
    }
    __syncthreads();
  }
#pragma unroll
  for (int i = 0; i < TM; ++i)
#pragma unroll
    for (int j = 0; j < TN; ++j) {
      const int r = bm0 + tr + i, c = bn0 + tc + j;
      if (r < M && c < N)
        C[r * ldc + c] = hx_out(acc[i][j], alpha, beta, beta == 0.f ? 0.f : C[r * ldc + c]);
    }
}

namespace {
bool aligned16(const void* p) { return (reinterpret_cast<uintptr_t>(p) & 15) == 0; }
int smem_optin() {
  int dev = 0, v = 0;
  cudaGetDevice(&dev);
  cudaDeviceGetAttribute(&v, cudaDevAttrMaxSharedMemoryPerBlockOptin, dev);
  return v;
}
template <int BM, int BN, int BK, int TM, int TN>
int run(int M, int N, int K, float alpha, const float* A, int lda, const float* B, int ldb,
        float beta, float* C, int ldc, cudaStream_t s) {
  constexpr int NT = (BM / TM) * (BN / TN);
  const int bytes = (BM * BK + BK * BN) * 4;
  if (NT < 64 || NT > 1024 || bytes > smem_optin()) return HXGEMM_ETILE;
  const bool vec = aligned16(A) && aligned16(B) && lda % 4 == 0 && ldb % 4 == 0;
  dim3 g((N + BN - 1) / BN, (M + BM - 1) / BM);
  if (vec) {
    auto k = k04_reg<BM, BN, BK, TM, TN, true>;
    cudaFuncSetAttribute(k, cudaFuncAttributeMaxDynamicSharedMemorySize, bytes);
    k<<<g, NT, bytes, s>>>(M, N, K, alpha, A, lda, B, ldb, beta, C, ldc);
  } else {
    auto k = k04_reg<BM, BN, BK, TM, TN, false>;
    cudaFuncSetAttribute(k, cudaFuncAttributeMaxDynamicSharedMemorySize, bytes);
    k<<<g, NT, bytes, s>>>(M, N, K, alpha, A, lda, B, ldb, beta, C, ldc);
  }
  return HXGEMM_OK;
}
}  // namespace

int launch_k04(hxgemm_tile_t t, int M, int N, int K, float alpha, const float* A, int lda,
               const float* B, int ldb, float beta, float* C, int ldc, cudaStream_t s) {
  if (t.bm == 0) t = hxgemm_tile_t{128, 128, 8, 8, 8, 0};
#define HX_CASE(BM, BN, BK, TM, TN)                                            \
  if (t.bm == BM && t.bn == BN && t.bk == BK && t.tm == TM && t.tn == TN)     \
    return run<BM, BN, BK, TM, TN>(M, N, K, alpha, A, lda, B, ldb, beta, C, ldc, s);
  HX_TILES(HX_CASE)
#undef HX_CASE
  return HXGEMM_ETILE;
}

int list_k04(hxgemm_tile_t* out, int cap) {
  int n = 0;
  const int lim = smem_optin();
#define HX_ADD(BM, BN, BK, TM, TN)                                                         \
  if ((BM / TM) * (BN / TN) >= 64 && (BM / TM) * (BN / TN) <= 1024 &&                     \
      (BM * BK + BK * BN) * 4 <= lim && n < cap) out[n++] = hxgemm_tile_t{BM, BN, BK, TM, TN, 0};
  HX_TILES(HX_ADD)
#undef HX_ADD
  return n;
}
```

```cpp
#pragma once
// src_bytes < size zero-fills the rest; src_bytes = 0 reads nothing.
__device__ __forceinline__ void cp_async16(void* smem, const void* gmem, int src_bytes) {
  unsigned s = static_cast<unsigned>(__cvta_generic_to_shared(smem));
  asm volatile("cp.async.cg.shared.global [%0], [%1], 16, %2;\n" ::"r"(s), "l"(gmem), "r"(src_bytes));
}
__device__ __forceinline__ void cp_async4(void* smem, const void* gmem, int src_bytes) {
  unsigned s = static_cast<unsigned>(__cvta_generic_to_shared(smem));
  asm volatile("cp.async.ca.shared.global [%0], [%1], 4, %2;\n" ::"r"(s), "l"(gmem), "r"(src_bytes));
}
__device__ __forceinline__ void cp_async_commit() { asm volatile("cp.async.commit_group;\n" ::); }
template <int N>
__device__ __forceinline__ void cp_async_wait() { asm volatile("cp.async.wait_group %0;\n" ::"n"(N)); }
```

`k05_double_buf.cu`의 커널: k04와 같은 계산 루프이되 A 타일을 **행 우선**(`As[BM][BK]`)으로 둔다(cp.async는 연속 바이트만 복사하므로 전치 저장이 안 됨). 내부 루프는 `ra[i] = As[(tr + i) * BK + kk]`. 스테이지 S개 버퍼를 dynamic shared에 연속 배치.

```cpp
template <int BM, int BN, int BK, int TM, int TN, int S, bool VEC>
__global__ void __launch_bounds__((BM / TM) * (BN / TN))
k05_pipe(int M, int N, int K, float alpha, const float* A, int lda, const float* B, int ldb,
         float beta, float* C, int ldc) {
  constexpr int NT = (BM / TM) * (BN / TN), SA = BM * BK, SB = BK * BN;
  extern __shared__ float smem[];
  const int tid = threadIdx.x, bm0 = blockIdx.y * BM, bn0 = blockIdx.x * BN;
  const int tr = (tid / (BN / TN)) * TM, tc = (tid % (BN / TN)) * TN;
  const int ntile = (K + BK - 1) / BK;
  auto load = [&](int t, int slot) {
    float* As = smem + slot * (SA + SB);
    float* Bs = As + SA;
    const int k0 = t * BK;
    if (VEC) {
      for (int i = tid; i < SA / 4; i += NT) {
        const int r = i / (BK / 4), c = (i % (BK / 4)) * 4, gr = bm0 + r, gc = k0 + c;
        const int n = (gr < M) ? max(0, min(4, K - gc)) : 0;
        cp_async16(&As[r * BK + c], n ? &A[gr * lda + gc] : A, n * 4);
      }
      for (int i = tid; i < SB / 4; i += NT) {
        const int r = i / (BN / 4), c = (i % (BN / 4)) * 4, gr = k0 + r, gc = bn0 + c;
        const int n = (gr < K) ? max(0, min(4, N - gc)) : 0;
        cp_async16(&Bs[r * BN + c], n ? &B[gr * ldb + gc] : B, n * 4);
      }
    } else {
      for (int i = tid; i < SA; i += NT) {
        const int r = i / BK, c = i % BK, ok = bm0 + r < M && k0 + c < K;
        cp_async4(&As[r * BK + c], ok ? &A[(bm0 + r) * lda + k0 + c] : A, ok ? 4 : 0);
      }
      for (int i = tid; i < SB; i += NT) {
        const int r = i / BN, c = i % BN, ok = k0 + r < K && bn0 + c < N;
        cp_async4(&Bs[r * BN + c], ok ? &B[(k0 + r) * ldb + bn0 + c] : B, ok ? 4 : 0);
      }
    }
  };
  float acc[TM][TN] = {};
  float ra[TM], rb[TN];
#pragma unroll
  for (int p = 0; p < S - 1; ++p) {  // prologue: tiles 0..S-2
    if (p < ntile) load(p, p);
    cp_async_commit();
  }
  for (int t = 0; t < ntile; ++t) {
    cp_async_wait<S - 2>();  // tile t has landed
    __syncthreads();
    const int nt = t + S - 1;  // refill the slot freed by tile t-1
    if (nt < ntile) load(nt, nt % S);
    cp_async_commit();
    const float* As = smem + (t % S) * (SA + SB);
    const float* Bs = As + SA;
#pragma unroll
    for (int kk = 0; kk < BK; ++kk) {
#pragma unroll
      for (int i = 0; i < TM; ++i) ra[i] = As[(tr + i) * BK + kk];
#pragma unroll
      for (int j = 0; j < TN; ++j) rb[j] = Bs[kk * BN + tc + j];
#pragma unroll
      for (int i = 0; i < TM; ++i)
#pragma unroll
        for (int j = 0; j < TN; ++j) acc[i][j] += ra[i] * rb[j];
    }
  }
#pragma unroll
  for (int i = 0; i < TM; ++i)
#pragma unroll
    for (int j = 0; j < TN; ++j) {
      const int r = bm0 + tr + i, c = bn0 + tc + j;
      if (r < M && c < N)
        C[r * ldc + c] = hx_out(acc[i][j], alpha, beta, beta == 0.f ? 0.f : C[r * ldc + c]);
    }
}
```

주의: 리필하는 슬롯은 t−1이 쓰던 슬롯이고, 그 슬롯을 읽던 계산은 이번 반복 처음의 `__syncthreads()` 이전에 끝났으므로 안전하다(S = 2일 때 슬롯 `(t+1) % 2`는 t−1의 슬롯). VEC 경로의 16 B 복사는 행 끝(`n < 4`)에서 `src_bytes = n*4`로 나머지를 0으로 채운다. 이는 `gc`가 4의 배수이고 `lda % 4 == 0`·포인터 16 B 정렬일 때만 안전하므로 VEC 조건은 k04와 같다.

`run`/`launch_k05`/`list_k05`는 k04와 같은 모양(템플릿 인자에 S 추가, shared 바이트 = `S*(SA+SB)*4`, `HX_TILES`를 S ∈ {2,3}로 두 번 펼침).

```python
import math
import pytest
import torch

import hxgemm_ext  # noqa: F401  registers torch.ops.hxgemm

torch.backends.cuda.matmul.allow_tf32 = False

KERNELS = [1, 2, 3, 4, 5]
SIZES = [1, 7, 33, 128, 1000, 2047, 4096]


def ref(a, b, c, alpha, beta):
    out = alpha * (a.double() @ b.double())
    if beta != 0:
        out += beta * c.double()
    return out


def close(got, want, k):
    rtol, atol = 1e-5 * math.sqrt(k), 1e-6 * k
    return torch.all((got.double() - want).abs() <= atol + rtol * want.abs())


@pytest.mark.parametrize("kernel", KERNELS)
@pytest.mark.parametrize("n", SIZES)
def test_square(kernel, n):
    if kernel <= 2 and n > 2047:
        pytest.skip("naive kernels are too slow at 4096")
    g = torch.Generator(device="cuda").manual_seed(0)
    a = torch.rand(n, n, device="cuda", generator=g) * 2 - 1
    b = torch.rand(n, n, device="cuda", generator=g) * 2 - 1
    got = torch.ops.hxgemm.sgemm(a, b, kernel, [])
    assert close(got, ref(a, b, None, 1.0, 0.0), n)


@pytest.mark.parametrize("kernel", KERNELS)
@pytest.mark.parametrize("alpha,beta", [(1.0, 0.0), (0.5, 2.0), (1.0, 1.0)])
def test_alpha_beta_and_offset(kernel, alpha, beta):
    m, n, k = 129, 67, 45
    big_a = torch.rand(m, k + 1, device="cuda") * 2 - 1
    big_b = torch.rand(k, n + 1, device="cuda") * 2 - 1
    a, b = big_a[:, 1:], big_b[:, 1:]   # pointer offset 1 float, ld = cols + 1
    c = torch.rand(m, n, device="cuda")
    want = ref(a, b, c, alpha, beta)
    torch.ops.hxgemm.sgemm_out(c, a, b, kernel, [], alpha, beta)
    assert close(c, want, k)


@pytest.mark.parametrize("kernel", KERNELS)
def test_beta_zero_ignores_nan(kernel):
    a = torch.rand(40, 50, device="cuda")
    b = torch.rand(50, 60, device="cuda")
    c = torch.full((40, 60), float("nan"), device="cuda")
    torch.ops.hxgemm.sgemm_out(c, a, b, kernel, [], 1.0, 0.0)
    assert not torch.isnan(c).any()


def test_rejects_bad_input():
    a = torch.rand(4, 5, device="cuda")
    with pytest.raises(ValueError):
        torch.ops.hxgemm.sgemm(a, torch.rand(6, 4, device="cuda"), 1, [])
    with pytest.raises(ValueError):
        torch.ops.hxgemm.sgemm(a.double(), torch.rand(5, 4, device="cuda").double(), 1, [])
    with pytest.raises(ValueError):
        torch.ops.hxgemm.sgemm(a.t(), torch.rand(4, 4, device="cuda"), 1, [])  # column-major
```

```cpp
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include "gemm.h"

namespace {
hxgemm_tile_t to_tile(c10::IntArrayRef t) {
  if (t.empty()) return {0, 0, 0, 0, 0, 0};
  TORCH_CHECK_VALUE(t.size() == 6, "tile must have 6 entries");
  return {int(t[0]), int(t[1]), int(t[2]), int(t[3]), int(t[4]), int(t[5])};
}
void check_mat(const at::Tensor& x, const char* name) {
  TORCH_CHECK_VALUE(x.is_cuda() && x.scalar_type() == at::kFloat && x.dim() == 2, name,
                    " must be a 2-D float32 CUDA tensor");
  TORCH_CHECK_VALUE(x.stride(1) == 1 && x.stride(0) >= x.size(1), name,
                    " must be row-major (unit column stride)");
}
void run(at::Tensor& c, const at::Tensor& a, const at::Tensor& b, int64_t kernel,
         c10::IntArrayRef tile, double alpha, double beta) {
  check_mat(a, "a"); check_mat(b, "b"); check_mat(c, "c");
  TORCH_CHECK_VALUE(a.size(1) == b.size(0) && c.size(0) == a.size(0) && c.size(1) == b.size(1),
                    "shape mismatch");
  int r = hxgemm_sgemm(static_cast<hxgemm_kernel_t>(kernel), to_tile(tile), a.size(0), b.size(1),
                       a.size(1), float(alpha), a.data_ptr<float>(), a.stride(0),
                       b.data_ptr<float>(), b.stride(0), float(beta), c.data_ptr<float>(),
                       c.stride(0), at::cuda::getCurrentCUDAStream());
  TORCH_CHECK_VALUE(r != HXGEMM_EINVAL && r != HXGEMM_ETILE, "hxgemm rejected the arguments (", r, ")");
  TORCH_CHECK(r == 0, "hxgemm CUDA error ", r);
}
at::Tensor sgemm(const at::Tensor& a, const at::Tensor& b, int64_t kernel, c10::IntArrayRef tile) {
  auto c = at::empty({a.size(0), b.size(1)}, a.options());
  run(c, a, b, kernel, tile, 1.0, 0.0);
  return c;
}
void sgemm_out(at::Tensor& c, const at::Tensor& a, const at::Tensor& b, int64_t kernel,
               c10::IntArrayRef tile, double alpha, double beta) {
  run(c, a, b, kernel, tile, alpha, beta);
}
}  // namespace

TORCH_LIBRARY(hxgemm, m) {
  m.def("sgemm(Tensor a, Tensor b, int kernel, int[] tile) -> Tensor");
  m.def("sgemm_out(Tensor(a!) c, Tensor a, Tensor b, int kernel, int[] tile, float alpha, float beta) -> ()");
}
TORCH_LIBRARY_IMPL(hxgemm, CUDA, m) {
  m.impl("sgemm", &sgemm);
  m.impl("sgemm_out", &sgemm_out);
}
PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {}
```
